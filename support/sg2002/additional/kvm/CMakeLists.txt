cmake_minimum_required(VERSION 3.16)

# Both builds install libkvm_mmf.so next to this wrapper.
set(CMAKE_BUILD_WITH_INSTALL_RPATH YES)
set(CMAKE_INSTALL_RPATH "\$ORIGIN")

if(COMMAND register_component)
    list(APPEND ADD_REQUIREMENTS kvm_mmf)
    list(APPEND ADD_INCLUDE "include")
    list(APPEND ADD_LINK_DEFINITIONS_PRIVATE -Wl,--as-needed)
    list(APPEND ADD_SRCS "src/kvm_vision.cpp")
    register_component(DYNAMIC)
    return()
endif()

project(kvm LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
set(KVM_MMF_PREFIX "" CACHE PATH "Installed matching kvm_mmf")
find_path(KVM_MMF_INCLUDE kvm_mmf.hpp PATHS "${KVM_MMF_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(KVM_MMF_LIBRARY kvm_mmf PATHS "${KVM_MMF_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(kvm SHARED src/kvm_vision.cpp)
target_compile_features(kvm PRIVATE cxx_std_17)
target_compile_options(kvm PRIVATE -Wall -Wextra)
target_include_directories(kvm PUBLIC include PRIVATE "${KVM_MMF_INCLUDE}")
target_link_libraries(kvm PRIVATE "${KVM_MMF_LIBRARY}" Threads::Threads atomic)
target_link_options(kvm PRIVATE -Wl,--as-needed -Wl,-z,defs -Wl,--no-allow-shlib-undefined)
install(TARGETS kvm LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/kvm_vision.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
